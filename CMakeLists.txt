cmake_minimum_required(VERSION 3.22)
project(ScreeningRoom VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Quick Network OpenGL WebEngineQuick WebEngineCore WebChannel WebSockets Sql)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
add_library(screening_mpv ALIAS PkgConfig::MPV)
add_executable(screening-room native/main.cpp native/player.cpp native/player.h native/videorenderer.cpp native/videorenderer.h native/library.cpp native/library.h native/tmdb.cpp native/tmdb.h native/cinepro.cpp native/cinepro.h)
target_sources(screening-room PRIVATE native/onlinefilter.cpp native/onlinefilter.h)
target_sources(screening-room PRIVATE native/networkremote.cpp native/networkremote.h)
target_link_libraries(screening-room PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Network Qt6::OpenGL Qt6::WebEngineQuick Qt6::WebEngineCore Qt6::WebChannel Qt6::WebSockets Qt6::Sql screening_mpv)
target_compile_options(screening-room PRIVATE -Wall -Wextra -Wpedantic)
option(SCREENING_ROOM_TESTING "Enable Qt input simulation for native integration tests" OFF)
if(SCREENING_ROOM_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  target_link_libraries(screening-room PRIVATE Qt6::Test)
  target_compile_definitions(screening-room PRIVATE SCREENING_ROOM_TESTING)
  add_executable(metadata-tests tests/metadata.cpp native/library.cpp native/library.h native/tmdb.cpp native/tmdb.h)
  target_include_directories(metadata-tests PRIVATE native)
  target_link_libraries(metadata-tests PRIVATE Qt6::Core Qt6::Network Qt6::Sql Qt6::Test)
  add_executable(online-filter-tests tests/online-filter.cpp native/onlinefilter.cpp native/onlinefilter.h)
  target_include_directories(online-filter-tests PRIVATE native)
  target_link_libraries(online-filter-tests PRIVATE Qt6::Core Qt6::WebEngineQuick Qt6::WebEngineCore Qt6::Test)
  qt_add_resources(online-filter-tests online_test_rules PREFIX / FILES config/online-filter.json)
  add_executable(network-remote-tests tests/network-remote.cpp native/networkremote.cpp native/networkremote.h)
  target_include_directories(network-remote-tests PRIVATE native)
  target_link_libraries(network-remote-tests PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Network Qt6::WebSockets Qt6::Test)
endif()
qt_add_resources(screening-room qml PREFIX / FILES native/Main.qml native/PlaybackControls.qml native/CineProLoading.qml native/quality-adapter.js src/assets/cinepro-loading.svg)
qt_add_resources(screening-room online_rules PREFIX / FILES config/online-filter.json)
file(GLOB_RECURSE WEB_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/dist/*")
if(NOT WEB_FILES)
  message(FATAL_ERROR "Run npm run build before configuring the native app")
endif()
qt_add_resources(screening-room web PREFIX /ui BASE "${CMAKE_CURRENT_SOURCE_DIR}/dist" FILES ${WEB_FILES})
install(TARGETS screening-room RUNTIME DESTINATION bin)

qt_add_resources(screening-room web_remote PREFIX / FILES public/remote/index.html public/remote/remote.css public/remote/remote.js)
if(SCREENING_ROOM_TESTING)
  qt_add_resources(network-remote-tests web_remote_test PREFIX / FILES public/remote/index.html public/remote/remote.css public/remote/remote.js)
endif()
